// 1. toggleSession 함수 수정 (sessionStartTime 여유를 0.15s로 확보하여 1번 박 씹힘 원천 차단) function toggleSession() { unlockAudioPipeline(); stopDemoPlay(); if (isRunning) return; beatInterval = 60 / bpm; const currLevel = CURRICULUM[currentLevelIdx]; const notesPerBar = currLevel.pattern.length; noteInterval = beatInterval / (notesPerBar === 16 ? 4 : 2); targetNotes = []; scheduledBeats = 0; currentCombo = 0; maxCombo = 0; // 0.15초 여유를 두어 AudioContext가 안정화된 뒤 1번 박부터 정확히 발음 sessionStartTime = audioCtx.currentTime + 0.15; playStartTime = sessionStartTime + (COUNT_IN_BEATS * beatInterval); nextBeatTime = sessionStartTime; for (let b = 0; b < TOTAL_BARS; b++) { for (let s = 0; s < notesPerBar; s++) { const hand = currLevel.pattern[s]; const stepIdx = (b * notesPerBar) + s; const targetTime = playStartTime + (stepIdx * noteInterval); if (hand === 'R' || hand === 'L') { targetNotes.push({ index: targetNotes.length, stepIndex: stepIdx, hand: hand, targetTime: targetTime, hit: false, judged: false, offsetMs: 0 }); } } } isRunning = true; document.getElementById('startBtn').style.display = 'none'; document.getElementById('resultModal').style.display = 'none'; document.getElementById('comboCounter').textContent = `0 COMBO`; scheduler(); } // 2. updateUIProgress 함수 수정 (1번 박이 울리기 전에는 준비 상태를 띄우고, 1,2,3,4박이 각각 정확한 타이밍에 갱신) function updateUIProgress(current) { if (current < playStartTime) { if (current < sessionStartTime) { document.getElementById('sessionStatus').innerHTML = `카운트다운 준비 중...`; } else { const count = Math.floor((current - sessionStartTime) / beatInterval) + 1; const displayCount = Math.min(COUNT_IN_BEATS, Math.max(1, count)); document.getElementById('sessionStatus').innerHTML = `예비박 카운트: ${displayCount} / ${COUNT_IN_BEATS}`; } } else { const elapsed = current - playStartTime; const currentStep = Math.floor(elapsed / noteInterval); const currentBar = Math.min(TOTAL_BARS, Math.floor(elapsed / (4 * beatInterval)) + 1); const subName = CURRICULUM[currentLevelIdx].pattern.length === 16 ? "16비트" : "8비트"; document.getElementById('sessionStatus').innerHTML = `🔥 ${subName} 연주 중 [ ${currentBar} / ${TOTAL_BARS} 마디 ]`; const patternLen = CURRICULUM[currentLevelIdx].pattern.length; const activeIdx = currentStep % patternLen; document.querySelectorAll('.step-node').forEach((node, idx) => { node.classList.toggle('active', idx === activeIdx); }); } } // 3. handleInput 함수 수정 (예비박 보호 범위를 noteInterval에 맞춰 동적으로 조정) function handleInput(hand, clientX, clientY) { createRipple(hand, clientX, clientY); if (!isRunning || !audioCtx) return; const current = audioCtx.currentTime; const compTime = current - 0.025; const maxWindow = noteInterval * 0.50; // 예비박 4박 동안 패드 오타격 방지 (단, 1번 본노트의 Early 어택은 정상 허용) if (current < playStartTime - maxWindow) { updateFeedback("예비박 카운트 중...", "var(--accent-orange)"); return; } for (let i = 0; i < targetNotes.length; i++) { const n = targetNotes[i]; if (!n.judged && (compTime - n.targetTime) > maxWindow) { n.judged = true; n.hit = false; n.offsetMs = 999; currentCombo = 0; } } const target = targetNotes.find(n => !n.judged); if (!target) return; const diff = compTime - target.targetTime; const diffMs = diff * 1000; const absMs = Math.abs(diffMs); if (absMs <= maxWindow * 1000) { target.judged = true; if (target.hand === hand) { target.hit = true; target.offsetMs = diffMs; const syncPercent = Math.max(55, Math.min(100, Math.round(100 - (absMs * 0.22)))); if (absMs <= 45) { currentCombo++; updateFeedback(`PERFECT (${syncPercent}% 싱크)`, "var(--accent-right)"); } else if (absMs <= 85) { currentCombo++; updateFeedback(`GREAT (${syncPercent}% 싱크)`, "var(--accent-left)"); } else { const tag = diffMs < 0 ? "EARLY" : "LATE"; updateFeedback(`${tag} (${syncPercent}% 싱크)`, "var(--accent-orange)"); currentCombo = 0; } } else { target.hit = false; target.offsetMs = 999; currentCombo = 0; updateFeedback("WRONG HAND!", "var(--accent-miss)"); } if (currentCombo > maxCombo) maxCombo = currentCombo; document.getElementById('comboCounter').textContent = `${currentCombo} COMBO`; } else if (diff < -maxWindow) { updateFeedback("TOO EARLY!", "var(--accent-orange)"); } }
